# Pinned to the versions the test suite is verified against. These were bumped
# from the original pins after `pip-audit` reported known vulnerabilities in
# PyJWT (which validates auth tokens) and Starlette (via FastAPI).
fastapi==0.140.0
uvicorn[standard]==0.51.0
pydantic==2.13.4
pydantic-settings==2.14.2
loguru==0.7.3
httpx==0.28.1
# Generated patches are applied to production clusters; hand-rolled YAML
# quoting is not a risk worth taking.
PyYAML==6.0.2
# Hand-rolling JWT signature verification is a well-known way to ship an auth
# bypass; use the maintained implementation.
PyJWT[crypto]==2.13.0
# X.509 for agent identity (app/security/): the CA that issues agent
# certificates, and the parsing that turns a peer certificate into a cluster.
# Already present transitively via PyJWT[crypto], but pinned here because M4b
# depends on it directly — a transitive pin is not a contract.
cryptography==50.0.0
# Wire contract for the cluster-agent protocol (/proto). The runtime version
# must match the grpcio-tools that generated app/wire/gen — protobuf 7 validates
# gencode against the runtime and fails loudly on a mismatch, which is why both
# are pinned and CI regenerates to prove they agree.
protobuf==7.35.1
# Durable job and report state. Imported only when DATABASE_URL and REDIS_URL
# are set, so the single-process deployment never loads either driver — but
# both are installed unconditionally so `pip install -r requirements.txt`
# stays the one instruction, and so the opt-in integration suite can run.
psycopg[binary,pool]==3.3.4
redis==8.1.0
# The agent transport. Imported only when AGENT_GATEWAY_PORT is set, so a
# deployment reading a local kubeconfig never loads it — but installed
# unconditionally so the generated service stubs import and `pip install -r
# requirements.txt` stays the one instruction.
grpcio==1.83.0

# Self-observability. The exposition format is trivial; correct histogram
# bucketing and a thread-safe registry are not, and a dashboard built on a
# quietly-wrong histogram is worse than no dashboard. `app/observability/`
# owns its own registry rather than the process-global default, so nothing
# else that imports this library can add series to /metrics.
prometheus-client==0.23.1
